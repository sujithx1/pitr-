Welcome to the PITR masterclass. This dashboard demonstrates how to safely archive PostgreSQL Write-Ahead Logs (WAL) using pgBackRest, analyze them in real-time to track database mutations, and perform exact millisecond-precision disaster recovery.

Production Configuration Files

Use these verified configuration files to set up your own production-ready Point-in-Time Recovery architecture.

docker-compose.yml 
services:
  postgres_pitr:
    build: ./postgres
    container_name: postgres_pitr_lab
    environment:
      POSTGRES_DB: db
      POSTGRES_USER: sujith
      POSTGRES_PASSWORD: sujith
    ports:
      - "5432:5432"
    volumes:
      - ./postgres/postgresql.conf:/etc/postgresql/postgresql.conf
      - ./postgres/pg_hba.conf:/etc/postgresql/pg_hba.conf
      - ./postgres/init.sql:/docker-entrypoint-initdb.d/init.sql
      - ./postgres/pgbackrest.conf:/etc/pgbackrest/pgbackrest.conf
      - ./backups:/backups
      - pgdata:/var/lib/postgresql
    command:
      [
        "postgres",
        "-c",
        "config_file=/etc/postgresql/postgresql.conf",
        "-c",
        "hba_file=/etc/postgresql/pg_hba.conf",
      ]

volumes:
  pgdata:
postgres/Dockerfile 
FROM postgres:18-alpine
RUN apk update && apk add pgbackrest
postgres/postgresql.conf 
# Connection Settings
listen_addresses = '*'
max_connections = 100

# Write-Ahead Log (WAL)
wal_level = replica                     # Essential for PITR
fsync = on                              
synchronous_commit = on                 

# Checkpoints
checkpoint_timeout = 5min               
max_wal_size = 1GB
min_wal_size = 80MB
checkpoint_completion_target = 0.9      

# Archiving
archive_mode = on                       
archive_command = 'pgbackrest --stanza=db archive-push %p'
postgres/pgbackrest.conf 
[db]
pg1-path=/var/lib/postgresql/18/docker
pg1-user=sujith

[global]
repo1-path=/backups
repo1-retention-full=2
log-level-console=info
log-level-file=detail
start-fast=y
compress-type=zst
